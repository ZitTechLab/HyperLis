; === Hàm ===
Func _UpdateStatus($msg)
    GUICtrlSetData($lblStatus, $msg)
EndFunc

Func _GetVersion()
    If GUICtrlRead($opt2019)=$GUI_CHECKED Then Return "Office 2019"
    If GUICtrlRead($opt2021)=$GUI_CHECKED Then Return "Office 2021"
    If GUICtrlRead($opt2024)=$GUI_CHECKED Then Return "Office 2024"
    If GUICtrlRead($opt365)=$GUI_CHECKED Then Return "Office 365"
    Return "Office 2024"
EndFunc

Func _GetProductID($ver)
    If StringInStr($ver, "2019") Then Return "ProPlus2019Retail"
    If StringInStr($ver, "2021") Then Return "ProPlus2021Retail"
    If StringInStr($ver, "2024") Then Return "ProPlus2024Retail"
    If StringInStr($ver, "365") Then Return "O365ProPlusRetail"
    Return "ProPlus2024Retail"
EndFunc

Func _GetChannel($ver)
    Return "Current"
EndFunc

Func _ExtractLangCode($text)
    Local $pos = StringInStr($text, "–")
    If $pos > 0 Then Return StringStripWS(StringTrimLeft($text, $pos+1), 3)
    Return "en-us"
EndFunc

Func _DownloadOffline($ver, $bitVer)
    Local $langText = GUICtrlRead($comboLang)
    Local $lang = _ExtractLangCode($langText)
    $bitVer = StringLeft($bitVer,2) ; "64-bit" => "64"
    Local $ProductID = _GetProductID($ver)
    Local $ChannelName = _GetChannel($ver)

    Local $sDownloadConfig = @TempDir & "\config_download.xml"
    Local $s = '<?xml version="1.0" encoding="UTF-8"?>' & @CRLF
    $s &= '<Configuration>' & @CRLF
    $s &= '  <Add OfficeClientEdition="' & $bitVer & '" Channel="' & $ChannelName & '" SourcePath="' & $sSourcePath & '">' & @CRLF
    $s &= '    <Product ID="' & $ProductID & '">' & @CRLF
    $s &= '      <Language ID="' & $lang & '" />' & @CRLF
    $s &= '    </Product>' & @CRLF
    $s &= '  </Add>' & @CRLF
    $s &= '</Configuration>'

    FileDelete($sDownloadConfig)
    FileWrite($sDownloadConfig, $s)

    _UpdateStatus("?? Ðang t?i: " & $ver & " (" & $bitVer & "-bit)...")
    Local $iPID = Run('"' & $sODT_Setup & '" /download "' & $sDownloadConfig & '"', "", @SW_HIDE)
    While ProcessExists($iPID)
        Sleep(1000)
    WEnd
    _UpdateStatus("? Ðã t?i xong: " & $ver & " (" & $bitVer & "-bit)")
EndFunc

Func _DownloadSelected()
    Local $ver = _GetVersion()
    Local $bitVer = GUICtrlRead($comboBit)
    _DownloadOffline($ver, $bitVer)
EndFunc

Func _DownloadAll()
    Local $versions[4]=["Office 2019","Office 2021","Office 2024","Office 365"]
    Local $bitVer = GUICtrlRead($comboBit)
    For $v In $versions
        _DownloadOffline($v, $bitVer)
    Next
    _UpdateStatus("? Ðã t?i xong t?t c?!")
EndFunc

Func _InstallOffice()
    Local $ver = _GetVersion()
    Local $bitVer = StringLeft(GUICtrlRead($comboBit),2)
    Local $langText = GUICtrlRead($comboLang)
    Local $lang = _ExtractLangCode($langText)

    Local $ProductID = _GetProductID($ver)
    Local $ChannelName = _GetChannel($ver)
    Local $ProjectID = "ProjectProRetail"
    Local $VisioID = "VisioProRetail"

    ; Ki?m tra thu vi?n offline
    If FileExists($sSourcePath) Then
        _UpdateStatus("?? Ðã tìm th?y thu vi?n offline, s? cài offline...")
    Else
        _UpdateStatus("?? Không có thu vi?n offline, s? cài online...")
        $sSourcePath = ""
    EndIf

    Local $sConfig = '<?xml version="1.0" encoding="UTF-8"?>' & @CRLF
    $sConfig &= '<Configuration>' & @CRLF
    $sConfig &= '  <Add OfficeClientEdition="' & $bitVer & '" Channel="' & $ChannelName & '"'
    If $sSourcePath <> "" Then
        $sConfig &= ' SourcePath="' & $sSourcePath & '"'
    EndIf
    $sConfig &= '>' & @CRLF

    $sConfig &= '    <Product ID="' & $ProductID & '">' & @CRLF
    $sConfig &= '      <Language ID="' & $lang & '" />' & @CRLF

    For $i=0 To UBound($aChkApp)-1
        If GUICtrlRead($aChkApp[$i])<>$GUI_CHECKED Then
            $sConfig &= '      <ExcludeApp ID="' & $aOfficeApps[$i] & '" />' & @CRLF
        EndIf
    Next

    $sConfig &= '    </Product>' & @CRLF

    If GUICtrlRead($chkProject)=$GUI_CHECKED Then
        $sConfig &= '    <Product ID="' & $ProjectID & '">' & @CRLF
        $sConfig &= '      <Language ID="' & $lang & '" />' & @CRLF
        $sConfig &= '    </Product>' & @CRLF
    EndIf
    If GUICtrlRead($chkVisio)=$GUI_CHECKED Then
        $sConfig &= '    <Product ID="' & $VisioID & '">' & @CRLF
        $sConfig &= '      <Language ID="' & $lang & '" />' & @CRLF
        $sConfig &= '    </Product>' & @CRLF
    EndIf

    $sConfig &= '  </Add>' & @CRLF
    $sConfig &= '</Configuration>'

    FileDelete($sConfigFile)
    FileWrite($sConfigFile, $sConfig)

    _UpdateStatus("?? Ðang cài d?t Office...")
    Local $iPID = Run('"' & $sODT_Setup & '" /configure "' & $sConfigFile & '"', "", @SW_HIDE)
    While ProcessExists($iPID)
        Sleep(1000)
    WEnd

    _UpdateStatus("? Ðã cài d?t xong!")
EndFunc
